name: Android APK and OTA

on:
  push:
    branches: [main]
  workflow_dispatch:

permissions:
  contents: write

concurrency:
  group: skyrim-android-main
  cancel-in-progress: false

jobs:
  android:
    runs-on: ubuntu-latest
    timeout-minutes: 40
    steps:
      - name: Checkout main
        uses: actions/checkout@v4
      - name: Install Java 17
        uses: actions/setup-java@v4
        with:
          distribution: temurin
          java-version: '17'
      - name: Install Gradle
        uses: gradle/actions/setup-gradle@v4
        with:
          gradle-version: '8.11.1'
      - name: Android SDK
        uses: android-actions/setup-android@v3
      - name: Install Android platform
        run: sdkmanager 'platforms;android-35' 'build-tools;35.0.0'
      - name: Assemble debug APK
        run: gradle --no-daemon --stacktrace :app:assembleDebug
      - name: Prepare debug artifact
        run: |
          mkdir -p dist
          cp app/build/outputs/apk/debug/app-debug.apk dist/SKYRIM-TEXT-RPG-debug.apk
      - name: Upload debug APK
        uses: actions/upload-artifact@v4
        with:
          name: SKYRIM-TEXT-RPG-debug
          path: dist/SKYRIM-TEXT-RPG-debug.apk
          if-no-files-found: error
          retention-days: 30
      - name: Detect release signing secret
        id: signing
        env:
          BUNDLE: ${{ secrets.SKYRIM_SIGNING_BUNDLE }}
        shell: bash
        run: |
          if [[ -n "$BUNDLE" ]]; then
            echo 'available=true' >> "$GITHUB_OUTPUT"
          else
            echo 'available=false' >> "$GITHUB_OUTPUT"
            echo 'Signed OTA release skipped: add SKYRIM_SIGNING_BUNDLE in repository Actions secrets.' >> "$GITHUB_STEP_SUMMARY"
          fi
      - name: Read version and check release tag
        if: steps.signing.outputs.available == 'true'
        id: release_version
        env:
          GH_TOKEN: ${{ github.token }}
        run: |
          VER=$(sed -n "s/^[[:space:]]*versionName '\([^']*\)'.*/\1/p" app/build.gradle | head -n1)
          test -n "$VER"
          echo "tag=v${VER}" >> "$GITHUB_OUTPUT"
          if gh release view "v${VER}" >/dev/null 2>&1; then
            echo "Version v${VER} already published. Bump app version before next OTA." >> "$GITHUB_STEP_SUMMARY"
            echo 'publish=false' >> "$GITHUB_OUTPUT"
          else
            echo 'publish=true' >> "$GITHUB_OUTPUT"
          fi
      - name: Unpack private signing key
        if: steps.signing.outputs.available == 'true' && steps.release_version.outputs.publish == 'true'
        env:
          BUNDLE: ${{ secrets.SKYRIM_SIGNING_BUNDLE }}
        run: |
          printf '%s' "$BUNDLE" | base64 -d > /tmp/skyrim-signing.zip
          unzip -qo /tmp/skyrim-signing.zip -d .
          test -s release.jks
          test -s signing.properties
          chmod 600 release.jks signing.properties
      - name: Assemble signed release APK
        if: steps.signing.outputs.available == 'true' && steps.release_version.outputs.publish == 'true'
        run: gradle --no-daemon --stacktrace :app:assembleRelease
      - name: Create update manifest and release notes
        if: steps.signing.outputs.available == 'true' && steps.release_version.outputs.publish == 'true'
        run: |
          cp app/build/outputs/apk/release/app-release.apk dist/SKYRIM-TEXT-RPG.apk
          python3 tools/build_manifest.py
      - name: Publish OTA release
        if: steps.signing.outputs.available == 'true' && steps.release_version.outputs.publish == 'true'
        uses: softprops/action-gh-release@v2
        with:
          tag_name: ${{ steps.release_version.outputs.tag }}
          name: SKYRIM TEXT RPG ${{ steps.release_version.outputs.tag }}
          body_path: dist/release_notes.md
          files: |
            dist/SKYRIM-TEXT-RPG.apk
            dist/update.json
          fail_on_unmatched_files: true
          draft: false
          prerelease: false
      - name: Upload signed APK artifact
        if: steps.signing.outputs.available == 'true' && steps.release_version.outputs.publish == 'true'
        uses: actions/upload-artifact@v4
        with:
          name: SKYRIM-TEXT-RPG-signed
          path: dist/SKYRIM-TEXT-RPG.apk
          if-no-files-found: error
          retention-days: 30
